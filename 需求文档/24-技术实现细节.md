### 24.1.1 坐标转换算法

**屏幕坐标到场景坐标**:

```typescript
function screenToScene(screenPoint: IPoint, viewport: Viewport): IPoint {
  const { centerX, centerY, zoom, offsetX, offsetY } = viewport;
  
  return {
    x: (screenPoint.x - offsetX) / zoom + centerX,
    y: (screenPoint.y - offsetY) / zoom + centerY,
  };
}
```

**场景坐标到屏幕坐标**:

```typescript
function sceneToScreen(scenePoint: IPoint, viewport: Viewport): IPoint {
  const { centerX, centerY, zoom, offsetX, offsetY } = viewport;
  
  return {
    x: (scenePoint.x - centerX) * zoom + offsetX,
    y: (scenePoint.y - centerY) * zoom + offsetY,
  };
}
```

### 24.1.2 矩阵变换算法

**变换矩阵应用**:

```typescript
function applyTransform(point: IPoint, matrix: IMatrixArr): IPoint {
  const [a, b, c, d, tx, ty] = matrix;
  
  return {
    x: a * point.x + c * point.y + tx,
    y: b * point.x + d * point.y + ty,
  };
}
```

**矩阵乘法**:

```typescript
function multiplyMatrix(m1: IMatrixArr, m2: IMatrixArr): IMatrixArr {
  const [a1, b1, c1, d1, tx1, ty1] = m1;
  const [a2, b2, c2, d2, tx2, ty2] = m2;
  
  return [
    a1 * a2 + c1 * b2,        // a
    b1 * a2 + d1 * b2,        // b
    a1 * c2 + c1 * d2,        // c
    b1 * c2 + d1 * d2,        // d
    a1 * tx2 + c1 * ty2 + tx1, // tx
    b1 * tx2 + d1 * ty2 + ty1, // ty
  ];
}
```

### 24.1.4 贝塞尔曲线算法

**二次贝塞尔曲线**:

```typescript
function quadraticBezier(t: number, p0: IPoint, p1: IPoint, p2: IPoint): IPoint {
  const u = 1 - t;
  const tt = t * t;
  const uu = u * u;
  
  return {
    x: uu * p0.x + 2 * u * t * p1.x + tt * p2.x,
    y: uu * p0.y + 2 * u * t * p1.y + tt * p2.y,
  };
}
```

**三次贝塞尔曲线**:

```typescript
function cubicBezier(t: number, p0: IPoint, p1: IPoint, p2: IPoint, p3: IPoint): IPoint {
  const u = 1 - t;
  const tt = t * t;
  const ttt = tt * t;
  const uu = u * u;
  const uuu = uu * u;
  
  return {
    x: uuu * p0.x + 3 * uu * t * p1.x + 3 * u * tt * p2.x + ttt * p3.x,
    y: uuu * p0.y + 3 * uu * t * p1.y + 3 * u * tt * p2.y + ttt * p3.y,
  };
}
```

**计算方法**:

```typescript
function calcBoundingBox(points: IPoint[]): IRect {
  if (points.length === 0) return { x: 0, y: 0, width: 0, height: 0 };
  
  let minX = Infinity, minY = Infinity;
  let maxX = -Infinity, maxY = -Infinity;
  
  for (const point of points) {
    minX = Math.min(minX, point.x);
    minY = Math.min(minY, point.y);
    maxX = Math.max(maxX, point.x);
    maxY = Math.max(maxY, point.y);
  }
  
  return {
    x: minX,
    y: minY,
    width: maxX - minX,
    height: maxY - minY,
  };
}
```

### 24.6.3 几何运算

**点到线距离**:

```typescript
function pointToLineDistance(point: IPoint, lineStart: IPoint, lineEnd: IPoint): number {
  const A = point.x - lineStart.x;
  const B = point.y - lineStart.y;
  const C = lineEnd.x - lineStart.x;
  const D = lineEnd.y - lineStart.y;
  
  const dot = A * C + B * D;
  const lenSq = C * C + D * D;
  const param = lenSq !== 0 ? dot / lenSq : -1;
  
  let xx, yy;
  if (param < 0) {
    xx = lineStart.x;
    yy = lineStart.y;
  } else if (param > 1) {
    xx = lineEnd.x;
    yy = lineEnd.y;
  } else {
    xx = lineStart.x + param * C;
    yy = lineStart.y + param * D;
  }
  
  const dx = point.x - xx;
  const dy = point.y - yy;
  return Math.sqrt(dx * dx + dy * dy);
}
```

## 24.7 性能优化技术

### 24.7.1 RAF 节流

**实现方式**:

```typescript
function rafThrottle<T extends (...args: any[]) => void>(fn: T): T {
  let rafId: number | null = null;
  let lastArgs: any[];
  
  return ((...args: any[]) => {
    lastArgs = args;
    
    if (rafId === null) {
      rafId = requestAnimationFrame(() => {
        fn(...lastArgs);
        rafId = null;
      });
    }
  }) as T;
}
```

### 24.7.2 对象池

**对象池实现**:

```typescript
class ObjectPool<T> {
  private pool: T[] = [];
  private createFn: () => T;
  
  constructor(createFn: () => T) {
    this.createFn = createFn;
  }
  
  get(): T {
    return this.pool.pop() || this.createFn();
  }
  
  release(obj: T): void {
    this.pool.push(obj);
  }
}
```

### 24.7.3 防抖和节流

**防抖**:

```typescript
function debounce<T extends (...args: any[]) => void>(
  fn: T, 
  delay: number
): T {
  let timeoutId: NodeJS.Timeout;
  
  return ((...args: any[]) => {
    clearTimeout(timeoutId);
    timeoutId = setTimeout(() => fn(...args), delay);
  }) as T;
}
```

**节流**:

```typescript
function throttle<T extends (...args: any[]) => void>(
  fn: T, 
  interval: number
): T {
  let lastCall = 0;
  
  return ((...args: any[]) => {
    const now = Date.now();
    if (now - lastCall >= interval) {
      lastCall = now;
      fn(...args);
    }
  }) as T;
}
```
